func TestFreshSnapshotPagesBudgetCountsPageOnceAndUnusedSlots(t *testing.T) {
	var pages core.ScannerSnapshotPages[diagnosticParserCoreVersionLexerSnapshot]
	var records [2]*diagnosticParserCoreVersionLexerSnapshot
	for i := range records {
		record, p, s, e, z, storage := pages.New([]byte{1}, nil, nil, make([]bool, 113))
		record.storage = storage
		record.dfa.externalPayload, record.dfa.externalTokenStart, record.dfa.externalTokenEnd, record.dfa.extZeroTried = p, s, e, z
		records[i] = record
	}
	model := func(multiple bool) *diagnosticParserCoreGenericScheduler {
		s := &diagnosticParserCoreGenericScheduler{versionLexerSnapshotPages: pages, footprintRefs: make([]diagnosticParserCoreFootprintRef, 0, 32), versionLexerRequests: make([]diagnosticParserCoreVersionLexerRequest, 2)}
		for i := range s.versionLexerRequests {
			record := records[0]
			if multiple {
				record = records[i]
			}
			s.versionLexerRequests[i].before = record
		}
		return s
	}
	one, many := model(false), model(true)
	if a, b := diagnosticParserCoreSchedulerFootprintBytes(one), diagnosticParserCoreSchedulerFootprintBytes(many); a != b {
		t.Fatalf("shared page charged per record: one=%d two=%d", a, b)
	}
	current, wide := pages.CurrentStorage()
	if current == nil || wide != nil {
		t.Fatal("wrong current storage")
	}
	empty := &diagnosticParserCoreGenericScheduler{footprintRefs: make([]diagnosticParserCoreFootprintRef, 0, 32)}
	before := diagnosticParserCoreSchedulerFootprintBytes(empty)
	empty.versionLexerSnapshotPages = pages
	if delta := diagnosticParserCoreSchedulerFootprintBytes(empty) - before; delta != current.FootprintBytes() {
		t.Fatalf("current page charge=%d want=%d", delta, current.FootprintBytes())
	}
	if direct := diagnosticParserCoreVersionLexerSnapshotFootprintBytes(records[0]); direct < current.FootprintBytes() {
		t.Fatal("standalone footprint omitted storage")
	}
}

func TestFreshSnapshotPagesKeepStaleVersionSnapshotImmutable(t *testing.T) {
	source, err := os.ReadFile("internal/benchfixtures/testdata/real/yaml")
	if err != nil {
		t.Fatal(err)
	}
	runner, err := newAdmissionCandidateRunner(NewParser(yamlSelectedProjectionLanguageForTest))
	if err != nil {
		t.Fatal(err)
	}
	tokens := runner.parser.acquireParserDFATokenSourceWithErrorRuns(source, false)
	if tokens == nil {
		t.Fatal("DFA source absent")
	}
	defer tokens.Close()
	dfa := tokens.snapshotRelexState()
	dfa.extZeroTried = make([]bool, len(runner.lang.ExternalSymbols))
	create := func() *diagnosticParserCoreVersionLexerSnapshot {
		checkpoint, err := runner.compact.InternCheckpoint(dfa.externalPayload)
		if err != nil {
			t.Fatal(err)
		}
		if err := runner.compact.SetPhaseCheckpoint(checkpoint); err != nil {
			t.Fatal(err)
		}
		if _, err := runner.compact.Seed(core.StateID(runner.lang.InitialState), 0); err != nil {
			t.Fatal(err)
		}
		runner.scheduler.compact = runner.compact
		runner.scheduler.tokenSource = tokens
		var result *diagnosticParserCoreVersionLexerSnapshot
		err = runner.compact.RunFreshSchedulerSession(func(owner core.SchedulerTransactionToken) error {
			var buildErr error
			result, buildErr = runner.scheduler.newVersionLexerSnapshot(owner, dfa, checkpoint, checkpoint)
			return buildErr
		})
		if err != nil {
			t.Fatal(err)
		}
		return result
	}
	retained := create()
	if retained.storage == nil {
		t.Fatalf("real scanner shape missed pages: payload=%d start=%d end=%d mask=%d", len(dfa.externalPayload), len(dfa.externalTokenStart), len(dfa.externalTokenEnd), len(dfa.extZeroTried))
	}
	data := bytes.Clone(retained.dfa.externalPayload)
	generation, storage := retained.coreGeneration, retained.storage
	if err := resetDiagnosticParserCoreGenericScheduler(&runner.scheduler); err != nil {
		t.Fatal(err)
	}
	if a, b := runner.scheduler.versionLexerSnapshotPages.CurrentStorage(); a != nil || b != nil {
		t.Fatal("scheduler retained a page after reset")
	}
	if err := runner.compact.Reset(); err != nil {
		t.Fatal(err)
	}
	next := create()
	if next == retained || next.storage == storage {
		t.Fatal("reset recycled a published address")
	}
	if retained.coreGeneration != generation || retained.storage != storage || !bytes.Equal(retained.dfa.externalPayload, data) {
		t.Fatal("reset overwrote a published snapshot")
	}
	if err := retained.validateDestination(runner.compact, runner.lang); err == nil {
		t.Fatal("stale snapshot regained authority")
	}
	clone := retained.clone()
	if clone.storage != nil {
		t.Fatal("independent clone still charged to original page")
	}
	if !bytes.Equal(clone.dfa.externalPayload, data) {
		t.Fatal("clone changed bytes")
	}
}

func TestSnapshotPairFootprintMultiplePagesInPointerOrder(t *testing.T) {
	var pages [8]core.ScannerSnapshotPages[diagnosticParserCoreVersionLexerSnapshot]
	var records []*diagnosticParserCoreVersionLexerSnapshot
	var expected uint64
	selected := 0
	var lowest uintptr = ^uintptr(0)
	for i := range pages {
		for j := 0; j < 2; j++ {
			record, p, s, e, z, storage := pages[i].New([]byte{byte(i)}, nil, nil, make([]bool, 113))
			record.storage = storage
			record.dfa.externalPayload, record.dfa.externalTokenStart, record.dfa.externalTokenEnd, record.dfa.extZeroTried = p, s, e, z
			records = append(records, record)
		}
		owner, _ := pages[i].CurrentStorage()
		expected += owner.FootprintBytes()
		if address := uintptr(unsafe.Pointer(owner)); address < lowest {
			lowest = address
			selected = i
		}
	}
	makeModel := func() *diagnosticParserCoreGenericScheduler {
		return &diagnosticParserCoreGenericScheduler{footprintRefs: make([]diagnosticParserCoreFootprintRef, 0, 256), versionLexerRequests: make([]diagnosticParserCoreVersionLexerRequest, len(records))}
	}
	empty, live := makeModel(), makeModel()
	live.versionLexerSnapshotPages = pages[selected]
	for i, record := range records {
		live.versionLexerRequests[len(records)-1-i].before = record
	}
	if delta := diagnosticParserCoreSchedulerFootprintBytes(live) - diagnosticParserCoreSchedulerFootprintBytes(empty); delta != expected {
		t.Fatalf("page accounting=%d want=%d", delta, expected)
	}
}
