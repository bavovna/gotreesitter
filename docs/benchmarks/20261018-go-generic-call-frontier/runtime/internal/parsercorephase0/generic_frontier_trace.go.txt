package parsercorephase0

import (
	"encoding/json"
	"fmt"
)

type genericTraceLeaf struct {
	ID                SubtreeID
	Symbol            Symbol
	Start, End        uint32
	External, Missing bool
}

func genericTraceLeaves(c *Core, root SubtreeID) []genericTraceLeaf {
	stack := []SubtreeID{root}
	var leaves []genericTraceLeaf
	for steps := 0; len(stack) != 0 && steps < 2048; steps++ {
		last := len(stack) - 1
		id := stack[last]
		stack = stack[:last]
		if id == 0 || uint64(id) > uint64(len(c.subtrees)) {
			panic("diagnostic trace invalid subtree")
		}
		r := c.subtrees[id-1]
		if r.terminal {
			leaves = append(leaves, genericTraceLeaf{id, r.symbol, r.startByte, r.endByte, r.external, r.missing})
			continue
		}
		for j := r.childCount; j > 0; j-- {
			child := c.children[r.firstChild+j-1]
			if child >= id {
				panic("diagnostic trace invalid child order")
			}
			stack = append(stack, child)
		}
	}
	if len(stack) != 0 {
		panic("diagnostic trace budget")
	}
	return leaves
}

func genericTracePrint(kind string, value any) {
	raw, err := json.Marshal(value)
	if err != nil {
		panic(err)
	}
	fmt.Printf("GENERIC_%s %s\n", kind, raw)
}

func traceGenericReconvergence(c *Core, left linkRecord, right linkInput) {
	before := c.Work()
	if left.prev == right.prev && c.reductionBatchContains(left.payload) && c.reductionBatchContains(right.payload) {
		l, r := c.subtrees[left.payload-1], c.subtrees[right.payload-1]
		genericTracePrint("RECONVERGENCE", struct {
			Enabled                         bool
			Left, Right                     SubtreeID
			Predecessor                     NodeID
			LeftProduction, RightProduction uint16
			LeftLeaves, RightLeaves         []genericTraceLeaf
		}{c.reductionScratch.selectCleanChildren, left.payload, right.payload, left.prev, l.productionID, r.productionID, genericTraceLeaves(c, left.payload), genericTraceLeaves(c, right.payload)})
	}
	if before != c.Work() {
		panic("read-only reconvergence trace changed work")
	}
}

func traceGenericLiveLinkCap(c *Core, key boundaryKey, in linkInput, oldID NodeID, links []linkRecord) {
	before := c.Work()
	type traceLink struct {
		Predecessor NodeID
		Payload     SubtreeID
		Score       int64
		Order       ForkOrder
		Leaves      []genericTraceLeaf
	}
	rows := make([]traceLink, 0, len(links)+1)
	for _, l := range links {
		rows = append(rows, traceLink{l.prev, l.payload, l.scoreDelta, ForkOrder{Value: l.order, Present: l.hasOrder()}, genericTraceLeaves(c, l.payload)})
	}
	rows = append(rows, traceLink{in.prev, in.payload, in.scoreDelta, in.order, genericTraceLeaves(c, in.payload)})
	genericTracePrint("CAP", struct {
		State StateID
		Byte  uint32
		Old   NodeID
		Links []traceLink
		Work  Work
	}{key.state, key.byteOffset, oldID, rows, before})
	if before != c.Work() {
		panic("read-only cap trace changed work")
	}
}
