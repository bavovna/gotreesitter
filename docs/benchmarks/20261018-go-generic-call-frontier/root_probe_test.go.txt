//go:build gts_parsercorephase0

package gotreesitter

import (
	"crypto/sha256"
	"fmt"
	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
	"testing"
)

func TestGoGenericCallFrontierProbe(t *testing.T) {
	cases := []struct{ name, source string }{
		{"single_identifier", "package p\nfunc f() { _ = F[int](x) }\n"},
		{"single_qualified", "package p\nfunc f() { _ = p.G[T](x) }\n"},
		{"combined", "package p\nfunc f() { _ = F[int](x); _ = p.G[T](x) }\n"},
		{"reverse", "package p\nfunc f() { _ = p.G[T](x); _ = F[int](x) }\n"},
		{"separate_functions", "package p\nfunc f() { _ = F[int](x) }; func g() { _ = p.G[T](x) }\n"},
		{"ordinary_calls", "package p\nfunc f() { _ = F(x); _ = p.G(x) }\n"},
	}
	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			type result struct {
				Error     string
				Core      core.Work
				Scheduler DiagnosticParserCoreGenericWork
				Tree      string
			}
			var results [2]result
			for mode := 0; mode < 2; mode++ {
				fmt.Printf("GENERIC_BEGIN case=%s observed=%d source_sha256=%x\n", tc.name, mode, sha256.Sum256([]byte(tc.source)))
				runner, err := newParserCoreFreshFullRunner(parserCoreWarmGoScanner, parserCoreFreshFullCanonicalOptions())
				if err != nil {
					t.Fatal(err)
				}
				observer := diagnosticParserCoreSeedObserver{}
				if mode == 1 {
					observer.afterElection = func(s *diagnosticParserCoreGenericScheduler) (bool, error) {
						fmt.Printf("GENERIC_ELECTION index=%d headers=%d symbol=%d start=%d end=%d\n", s.electionIndex, len(s.headers), s.token.Symbol, s.token.StartByte, s.token.EndByte)
						return false, nil
					}
				}
				s, tokens, err := runner.executeSchedulerOpenWithObserver([]byte(tc.source), runner.compact, false, observer)
				if err != nil {
					results[mode].Error = err.Error()
				} else {
					results[mode].Core = runner.compact.Work()
					results[mode].Scheduler = s.work
					tree, e := runner.materialize([]byte(tc.source), runner.compact, s.acceptedHead)
					if e != nil {
						t.Fatal(e)
					}
					if tree.RootNode().HasError() || tree.RootNode().EndByte() != uint32(len(tc.source)) {
						t.Fatal("incomplete witness")
					}
					results[mode].Tree = tree.RootNode().SExpr(runner.lang)
					tree.Release()
				}
				if tokens != nil {
					tokens.Close()
				}
				fmt.Printf("GENERIC_END case=%s observed=%d error=%q tree=%q\n", tc.name, mode, results[mode].Error, results[mode].Tree)
			}
			if results[0] != results[1] {
				t.Fatal("observer changed admission, committed work or tree")
			}
		})
	}
}
