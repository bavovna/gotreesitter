//go:build gts_parsercorephase0

package gotreesitter

import (
	"fmt"
	"reflect"

	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
)

type SharedOriginHeader struct {
	Sequence                                        uint64
	State                                           core.StateID
	Byte                                            uint32
	Checkpoint                                      core.CheckpointID
	Shifted, Paused, Accepted, ResurrectionUnproved bool
}

type SharedOriginEvent struct {
	Kind                 string
	Election             int
	Parent, Child        uint64
	TokenSymbol          Symbol
	TokenStart, TokenEnd uint32
	LexerState           StateID
	LexerByte            int
	States               []StateID
	Indices              []int
	Headers              []SharedOriginHeader
	Work                 DiagnosticParserCoreGenericWork
	CoreWork             core.Work
}

var diagnosticSharedOriginObserver func(*diagnosticParserCoreGenericScheduler, string, []int, uint64, uint64) error

func diagnosticSharedOriginRecord(s *diagnosticParserCoreGenericScheduler, kind string, indices []int, parent, child uint64) error {
	if diagnosticSharedOriginObserver == nil {
		return nil
	}
	return diagnosticSharedOriginObserver(s, kind, indices, parent, child)
}

func CaptureSharedLookaheadOriginForTest(detailed bool) func() []SharedOriginEvent {
	prior := diagnosticSharedOriginObserver
	var events []SharedOriginEvent
	diagnosticSharedOriginObserver = func(s *diagnosticParserCoreGenericScheduler, kind string, indices []int, parent, child uint64) error {
		if (!detailed && kind != "drop") || s.electionIndex > 32 {
			return nil
		}
		beforeWork, beforeCore := s.work, s.compact.Work()
		beforeHeaders := append([]diagnosticParserCoreHeader(nil), s.headers...)
		row := SharedOriginEvent{Kind: kind, Election: s.electionIndex, Parent: parent, Child: child, TokenSymbol: s.token.Symbol, TokenStart: s.token.StartByte, TokenEnd: s.token.EndByte, LexerState: s.tokenSource.state, LexerByte: s.tokenSource.lexer.pos, States: append([]StateID(nil), s.currentElection.States...), Indices: append([]int(nil), indices...), Work: s.work, CoreWork: s.compact.Work()}
		if kind == "before_lex" {
			row.Election++
			row.States = append([]StateID(nil), s.electStates...)
		}
		for _, h := range s.headers {
			state, offset, err := s.compact.Boundary(h.head)
			if err != nil {
				return err
			}
			row.Headers = append(row.Headers, SharedOriginHeader{Sequence: h.creationSeq, State: state, Byte: offset, Checkpoint: h.checkpoint, Shifted: h.shifted, Paused: h.paused, Accepted: h.accepted, ResurrectionUnproved: h.resurrectionUnproved})
		}
		if !reflect.DeepEqual(beforeWork, s.work) || !reflect.DeepEqual(beforeCore, s.compact.Work()) || !reflect.DeepEqual(beforeHeaders, s.headers) {
			return fmt.Errorf("shared origin observer changed scheduler state")
		}
		events = append(events, row)
		return nil
	}
	return func() []SharedOriginEvent { diagnosticSharedOriginObserver = prior; return events }
}
