//go:build gts_parsercorephase0

package gotreesitter_test

import (
	"bytes"
	"encoding/json"
	"os"
	"reflect"
	"testing"

	ts "github.com/odvcencio/gotreesitter"
	"github.com/odvcencio/gotreesitter/grammars"
	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
)

func TestYAMLSharedLookaheadOriginTrace(t *testing.T) {
	sample, err := os.ReadFile("internal/benchfixtures/testdata/real/yaml")
	if err != nil {
		t.Fatal(err)
	}
	doc := append([]byte("---\n"), sample...)
	doc = append(doc, '\n')
	for _, f := range []struct {
		name   string
		source []byte
	}{{"r4", sample}, {"four_documents", bytes.Repeat(doc, 4)}, {"sixteen_documents", bytes.Repeat(doc, 16)}} {
		t.Run(f.name, func(t *testing.T) {
			restore := core.SetExternalLexerOwnershipProbeForTest(false)
			defer restore()
			lang := grammars.YamlLanguage()
			p := ts.NewParser(lang)
			p.SetAdmissionCandidateRoute(true)
			type result struct {
				sexpr           string
				end             uint32
				hasError        bool
				tokens, nodes   uint64
				route, fallback uint64
			}
			parse := func() result {
				ts.ResetAdmissionCandidateCountersForTest()
				tree, err := p.Parse(f.source)
				if err != nil || tree == nil {
					t.Fatal(err)
				}
				defer tree.Release()
				r := tree.ParseRuntime()
				a, b := ts.AdmissionCandidateCounters()
				return result{tree.RootNode().SExpr(lang), tree.RootNode().EndByte(), tree.RootNode().HasError(), uint64(r.TokensConsumed), uint64(r.NodesAllocated), a, b}
			}
			plain := parse()
			finish := ts.CaptureSharedLookaheadOriginForTest(false)
			minimal := parse()
			reference := finish()
			finish = ts.CaptureSharedLookaheadOriginForTest(true)
			observed := parse()
			events := finish()
			if plain != minimal || plain != observed {
				t.Fatal("origin observer changed tree, route or token/node work")
			}
			var drops []ts.SharedOriginEvent
			for _, e := range events {
				if e.Kind == "drop" {
					drops = append(drops, e)
				}
			}
			if len(drops) == 0 || !reflect.DeepEqual(reference, drops) {
				t.Fatal("detailed observer changed complete drop-time work or headers")
			}
			data := struct {
				Fixture                         string
				Bytes                           int
				ObservedRoute, ObservedFallback uint64
				Events                          []ts.SharedOriginEvent
			}{f.name, len(f.source), observed.route, observed.fallback, events}
			raw, err := json.Marshal(data)
			if err != nil {
				t.Fatal(err)
			}
			t.Logf("SHARED_ORIGIN_TRACE %s", raw)
		})
	}
}
