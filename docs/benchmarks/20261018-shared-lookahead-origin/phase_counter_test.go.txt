//go:build gts_parsercorephase0

package gotreesitter_test

import (
	"bytes"
	"encoding/json"
	gts "github.com/odvcencio/gotreesitter"
	"github.com/odvcencio/gotreesitter/grammars"
	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
	"os"
	"testing"
)

func TestActualSharedRequestPhaseCounters(t *testing.T) {
	sample, err := os.ReadFile("internal/benchfixtures/testdata/real/yaml")
	if err != nil {
		t.Fatal(err)
	}
	doc := append([]byte("---\n"), sample...)
	doc = append(doc, '\n')
	for _, f := range []struct {
		name   string
		source []byte
	}{{"r4", sample}, {"four_documents", bytes.Repeat(doc, 4)}, {"sixteen_documents", bytes.Repeat(doc, 16)}} {
		for _, role := range []string{"ordinary", "owned", "actual"} {
			t.Run(f.name+"/"+role, func(t *testing.T) {
				restore := core.SetExternalLexerOwnershipProbeForTest(role == "owned")
				defer restore()
				var finish func() []gts.ActualSharedRequestObservation
				if role == "actual" {
					finish = gts.SeedActualSharedRequestsForTest()
				}
				p := gts.NewParser(grammars.YamlLanguage())
				p.SetAdmissionCandidateRoute(true)
				gts.ResetAdmissionCandidateCountersForTest()
				getWork := gts.CaptureActualSharedWorkForTest()
				tree, err := p.Parse(f.source)
				if err != nil || tree == nil {
					t.Fatal(err)
				}
				defer tree.Release()
				var observations []gts.ActualSharedRequestObservation
				if finish != nil {
					observations = finish()
				}
				routed, fallback := gts.AdmissionCandidateCounters()
				w, c, ok := getWork()
				activations := 0
				for _, e := range observations {
					if e.Stage == "" && !e.Drop {
						activations++
					}
				}
				data, err := json.Marshal(struct {
					Fixture, Role    string
					Bytes            int
					Routed, Fallback uint64
					Runtime          gts.ParseRuntime
					Work             gts.DiagnosticParserCoreGenericWork
					CoreWork         core.Work
					WorkAvailable    bool
					Activations      int
				}{f.name, role, len(f.source), routed, fallback, tree.ParseRuntime(), w, c, ok, activations})
				if err != nil {
					t.Fatal(err)
				}
				t.Logf("PHASE_COUNTERS %s", data)
			})
		}
	}
}
