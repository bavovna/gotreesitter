//go:build cgo && treesitter_c_parity && gts_parsercorephase0 && !gts_no_parsercorephase0

package cgoharness

import (
	"bytes"
	"fmt"
	"os"
	"testing"

	gts "github.com/odvcencio/gotreesitter"
	"github.com/odvcencio/gotreesitter/grammars"
	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
)

func TestActualSharedRequestLockedC(t *testing.T) {
	t.Cleanup(core.SetExternalLexerOwnershipProbeForTest(false))
	sample, err := os.ReadFile("../internal/benchfixtures/testdata/real/yaml")
	if err != nil {
		t.Fatal(err)
	}
	doc := append([]byte("---\n"), sample...)
	doc = append(doc, '\n')
	fixtures := []struct {
		name   string
		source []byte
	}{
		{"r4", sample},
		{"four_documents", bytes.Repeat(doc, 4)},
		{"sixteen_documents", bytes.Repeat(doc, 16)},
	}
	cursors := []string{
		"key: value\n", "key: value\r\nnext: ok\r\n", "key:\tvalue\n",
		"# comment with : and #\nkey: value # tail\n",
		"key: \"value: # \\t end\"\n", "key: 'value it''s quoted'\n",
		"key: |+\n  value\n  another line\n\n", "key: >-\n  value\n  continued\n",
		"[value, {nested: [one, two]}, three]\n", "{key: value, empty: null}\n",
		"key: &anchor value\nref: *anchor\n", "key: !!str value\n",
		"key: value café Ελληνικά 日本語 😀\n", "---\nkey: value\n...\n---\nnext: other\n",
		"key:\n  - value\n  - nested:\n      child: other\n", "key: value",
	}
	for i, source := range cursors {
		fixtures = append(fixtures, struct {
			name   string
			source []byte
		}{fmt.Sprintf("cursor_%02d", i), []byte(source)})
	}
	cLang, err := ParityCLanguage("yaml")
	if err != nil {
		t.Fatal(err)
	}
	lang := grammars.YamlLanguage()
	for _, f := range fixtures {
		t.Run(f.name, func(t *testing.T) {
			finish := gts.SeedActualSharedRequestsForTest()
			defer func() {
				events := finish()
				for _, e := range events {
					if e.ArtifactOption {
						t.Error("artifact exception used")
					}
				}
				t.Logf("request_observations=%d", len(events))
				for _, e := range events {
					if e.Stage == "pending_guard_coverage" {
						t.Logf("pending_guard_hits=%d shifted_hits=%d", e.Mask, e.Start)
					}
				}
			}()
			parse := func(p *gts.Parser, source []byte) *gts.Tree {
				t.Helper()
				beforeA, beforeB := gts.AdmissionCandidateCounters()
				tree, err := p.Parse(source)
				if err != nil || tree == nil {
					t.Fatalf("parse: %v", err)
				}
				afterA, afterB := gts.AdmissionCandidateCounters()
				// The stream witnesses must complete in compact. Some small
				// cursor cases may retain independent, documented fallbacks.
				if f.name == "r4" || f.name == "four_documents" || f.name == "sixteen_documents" {
					if afterA-beforeA != 1 || afterB-beforeB != 0 {
						t.Fatalf("compact=%d fallback=%d: %s", afterA-beforeA, afterB-beforeB, gts.AdmissionCandidateLastFallbackReason())
					}
				}
				ct := compactT3ParseC(t, cLang, source)
				defer ct.Close()
				assertLockedCTreeExactWithErrors(t, "fresh", tree, lang, ct)
				return tree
			}
			p := gts.NewParser(lang)
			p.SetAdmissionCandidateRoute(true)
			tree := parse(p, f.source)
			at := bytes.Index(f.source, []byte("value"))
			if at < 0 {
				at = bytes.Index(f.source, []byte("gemma"))
			}
			if at < 0 {
				t.Fatal("no edit marker")
			}
			edited := bytes.Clone(f.source)
			edited[at] = 'w'
			point := gts.Point{Row: uint32(bytes.Count(f.source[:at], []byte{'\n'})), Column: uint32(at - bytes.LastIndexByte(f.source[:at], '\n') - 1)}
			end := point
			end.Column++
			tree.Edit(gts.InputEdit{StartByte: uint32(at), OldEndByte: uint32(at + 1), NewEndByte: uint32(at + 1), StartPoint: point, OldEndPoint: end, NewEndPoint: end})
			incremental, err := p.ParseIncremental(edited, tree)
			tree.Release()
			if err != nil || incremental == nil {
				t.Fatalf("incremental: %v", err)
			}
			defer incremental.Release()
			ct := compactT3ParseC(t, cLang, edited)
			defer ct.Close()
			assertLockedCTreeExactWithErrors(t, "incremental", incremental, lang, ct)
			freshParser := gts.NewParser(lang)
			freshParser.SetAdmissionCandidateRoute(true)
			fresh := parse(freshParser, edited)
			defer fresh.Release()
			assertLockedCTreeExactWithErrors(t, "fresh after edit", fresh, lang, ct)
			var noedit *gts.Tree
			allocs := testing.AllocsPerRun(10, func() { noedit, err = p.ParseIncremental(edited, incremental) })
			if err != nil || noedit != incremental || allocs != 0 {
				t.Fatalf("no edit: same=%t allocs=%v err=%v", noedit == incremental, allocs, err)
			}
		})
	}
}
