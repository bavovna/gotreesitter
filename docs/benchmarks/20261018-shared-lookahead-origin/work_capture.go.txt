package gotreesitter

import core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"

var diagnosticSharedWorkObserver func(*diagnosticParserCoreGenericScheduler)

func diagnosticSharedWorkRecord(s *diagnosticParserCoreGenericScheduler) {
	if diagnosticSharedWorkObserver != nil {
		diagnosticSharedWorkObserver(s)
	}
}

// Capture work while the scheduler is still live. Pool return resets it.
func CaptureActualSharedWorkForTest() func() (DiagnosticParserCoreGenericWork, core.Work, bool) {
	prior := diagnosticSharedWorkObserver
	var w DiagnosticParserCoreGenericWork
	var c core.Work
	ok := false
	diagnosticSharedWorkObserver = func(s *diagnosticParserCoreGenericScheduler) { w = s.work; c = s.compact.Work(); ok = true }
	return func() (DiagnosticParserCoreGenericWork, core.Work, bool) {
		diagnosticSharedWorkObserver = prior
		return w, c, ok
	}
}
