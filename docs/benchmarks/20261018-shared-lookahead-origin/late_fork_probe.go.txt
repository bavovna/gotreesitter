//go:build gts_parsercorephase0

package gotreesitter

import (
	"errors"
	"reflect"

	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
)

var diagnosticSharedOriginObserver func(*diagnosticParserCoreGenericScheduler, string, []int, uint64, uint64) error

func diagnosticSharedOriginRecord(s *diagnosticParserCoreGenericScheduler, kind string, indices []int, parent, child uint64) error {
	if diagnosticSharedOriginObserver == nil {
		return nil
	}
	return diagnosticSharedOriginObserver(s, kind, indices, parent, child)
}

type ActualSharedRequestObservation struct {
	Stage          string
	Mask           uint32
	Election       int
	Owner          uint64
	OriginState    StateID
	Symbol         Symbol
	Start, End     uint32
	Drop           bool
	VersionProof   bool
	ArtifactOption bool
}

// SeedActualSharedRequestsForTest is a private experiment. The request is
// produced by the actual one-header shared Next, not inferred from mode
// equality or produced by lexing again after the header changes state.
func SeedActualSharedRequestsForTest() func() []ActualSharedRequestObservation {
	prior := diagnosticSharedOriginObserver
	var source *diagnosticParserCoreGenericScheduler
	var head core.Head
	var afterScratch dfaRelexSnapshotScratch
	var actualAfter dfaRelexSnapshot
	var actualToken Token
	var actualElection int
	var sequence, generation uint64
	var state StateID
	var contract diagnosticParserCoreVersionLexerScannerContract
	var identity [32]byte
	var identityRequired, identityValid bool
	valid := false
	var observations []ActualSharedRequestObservation
	diagnosticSharedOriginObserver = func(s *diagnosticParserCoreGenericScheduler, kind string, indices []int, _, _ uint64) error {
		if kind == "owned_classification" && s.electionIndex >= 29 && len(indices) == 2 {
			h := &s.headers[indices[0]]
			st, _, err := s.compact.Boundary(h.head)
			if err != nil {
				return err
			}
			observations = append(observations, ActualSharedRequestObservation{Stage: "classify", Election: s.electionIndex, Owner: h.creationSeq, OriginState: StateID(st), Mask: uint32(indices[1]), Symbol: s.token.Symbol, Start: s.token.StartByte, End: s.token.EndByte})
			return nil
		}
		if kind == "before_lex" {
			valid = false
			var mask uint32
			if s.versionLexerOwnershipActive {
				mask |= 1
			}
			if len(s.headers) != 1 {
				mask |= 2
			}
			if s.fullReceipts() {
				mask |= 4
			}
			if s.recoveryTurns.active {
				mask |= 8
			}
			if s.recoveryIsolation {
				mask |= 16
			}
			if !s.versionLexerBeforeValid {
				mask |= 32
			}
			if mask != 0 {
				observations = append(observations, ActualSharedRequestObservation{Stage: "before", Election: s.electionIndex + 1, OriginState: s.tokenSource.state, Mask: mask})
				return nil
			}
			h := &s.headers[0]
			if h.accepted || h.recoveryRegion() != nil || h.isRecoveryLineage() {
				return nil
			}
			var err error
			contract, err = diagnosticParserCoreVersionLexerScannerContractForLanguage(s.tokenSource.language)
			if err != nil || !contract.present {
				return nil
			}
			identity, identityRequired, identityValid = diagnosticParserCoreVersionLexerCheckpointIdentity(s.tokenSource.language)
			if identityRequired && !identityValid {
				return nil
			}
			source = s
			head = h.head
			sequence = h.creationSeq
			generation = s.compact.ResetGeneration()
			state = s.tokenSource.state
			valid = true
			return nil
		}
		if kind == "drop" {
			observations = append(observations, ActualSharedRequestObservation{Election: s.electionIndex, Drop: true, VersionProof: s.versionLexerNoActionProof, ArtifactOption: s.options.allowConvergedSplitDropArtifact})
			return nil
		}
		if kind == "after_lex" && valid {
			if s != source || len(s.headers) != 1 || s.versionLexerOwnershipActive || generation != s.compact.ResetGeneration() || len(s.currentElection.States) != 1 || s.currentElection.States[0] != state {
				valid = false
				return nil
			}
			h := &s.headers[0]
			token := s.token
			if h.head != head || h.creationSeq != sequence || h.shifted || h.accepted || h.paused || !token.ExternalScannerToken || token.Missing || token.NoLookahead || token.EndByte < token.StartByte || !reflect.DeepEqual(token, s.currentElection.Token) || s.tokenSource.lexer.pos != int(token.EndByte) {
				valid = false
				return nil
			}
			// This separate scratch is not borrowed from the state-specific
			// relex helpers. It remains frozen until the next shared election.
			actualAfter = s.tokenSource.snapshotRelexStateWithScratch(&afterScratch)
			actualToken = token
			actualElection = s.electionIndex
			return nil
		}
		if kind != "before_reduction_fork" || !valid {
			return nil
		}
		valid = false
		if len(indices) != 2 || indices[0] != 0 || indices[1] < 2 || s != source || len(s.headers) != 1 || s.versionLexerOwnershipActive || generation != s.compact.ResetGeneration() || s.electionIndex != actualElection || s.freshSessionOwner == nil {
			return nil
		}
		h := &s.headers[0]
		if h.creationSeq != sequence || h.shifted || h.accepted || h.paused || h.recoveryRegion() != nil || h.isRecoveryLineage() || !reflect.DeepEqual(s.token, actualToken) {
			return nil
		}
		if !core.ExternalLexerOwnsAllTerminals(s.tokenSource.language.TokenCount, s.tokenSource.language.ExternalSymbols) {
			return nil
		}
		token := actualToken
		afterDFA := actualAfter
		if err := s.seedVersionLexerOwnership(); err != nil {
			return err
		}
		before := s.headers[0].versionLexerSnapshot()
		if before == nil || before.metadata.Value().scanner != contract || before.metadata.Value().checkpointIdentity != identity || before.metadata.Value().checkpointIdentityRequired != identityRequired || before.metadata.Value().checkpointIdentityValid != identityValid {
			return errors.New("actual shared request descriptor changed across Next")
		}
		var after *diagnosticParserCoreVersionLexerSnapshot
		if err := s.withVersionLexerOwner(func(owner core.SchedulerTransactionToken) error {
			var err error
			after, err = s.newVersionLexerSnapshot(owner, afterDFA, s.checkpointID, s.checkpointID)
			return err
		}); err != nil {
			return err
		}
		if after.metadata.Value().scanner != contract || after.metadata.Value().checkpointIdentity != identity || after.metadata.Value().checkpointIdentityRequired != identityRequired || after.metadata.Value().checkpointIdentityValid != identityValid {
			return errors.New("actual shared after request descriptor changed")
		}
		request := diagnosticParserCoreVersionLexerRequest{electionIndex: s.electionIndex, headerCreationSeq: sequence, state: state, token: token, before: before, after: after, beforeCheckpoint: before.afterCheckpointInfo, afterCheckpoint: after.afterCheckpointInfo, beforeID: s.checkpointBeforeID, afterID: s.checkpointID, valid: true}
		s.versionLexerRequests = append(s.versionLexerRequests, request)
		ref := uint32(len(s.versionLexerRequests))
		if err := s.withVersionLexerOwner(func(owner core.SchedulerTransactionToken) error {
			return s.publishVersionLexerRequestOwned(owner, 0, before, ref)
		}); err != nil {
			return err
		}
		s.headers[0].checkpoint = request.afterID
		s.work.add(&s.work.PerVersionLexPublications, 1)
		s.work.add(&s.work.PerVersionLexRequests, 1)
		s.versionLexerOwnershipActive = true
		if uint64(len(s.headers)) > s.work.PeakLiveVersions {
			s.work.PeakLiveVersions = uint64(len(s.headers))
		}
		observations = append(observations, ActualSharedRequestObservation{Election: s.electionIndex, Owner: sequence, OriginState: state, Symbol: token.Symbol, Start: token.StartByte, End: token.EndByte})
		return nil
	}
	return func() []ActualSharedRequestObservation { diagnosticSharedOriginObserver = prior; return observations }
}
