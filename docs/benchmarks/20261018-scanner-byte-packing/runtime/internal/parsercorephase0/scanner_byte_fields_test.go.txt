package parsercorephase0

import (
	"bytes"
	"testing"
)

func TestCloneScannerByteFieldsIsolation(t *testing.T) {
	source := []byte("abcdef")
	p, s, e := CloneScannerByteFields(source[:3], source[1:4], source[2:5])
	for _, f := range [][]byte{p, s, e} {
		if len(f) != cap(f) {
			t.Fatal("field can append into a neighbor")
		}
	}
	source[1] = 'X'
	if !bytes.Equal(p, []byte("abc")) || !bytes.Equal(s, []byte("bcd")) || !bytes.Equal(e, []byte("cde")) {
		t.Fatal("snapshot aliases caller storage")
	}
	p[1] = 'Y'
	if string(s) != "bcd" || string(e) != "cde" {
		t.Fatal("private fields overlap")
	}
	p = append(p, 'Z')
	s[0] = 'Q'
	if string(p) != "aYcZ" || string(e) != "cde" {
		t.Fatal("field append or write changed another field")
	}
	other, _, _ := CloneScannerByteFields([]byte("abc"), nil, nil)
	other[0] = 'W'
	if string(p) != "aYcZ" {
		t.Fatal("separate freezes alias")
	}
}
func TestCloneScannerByteFieldsAbsentAndEmpty(t *testing.T) {
	for mask := 0; mask < 8; mask++ {
		fields := [3][]byte{}
		for i := range fields {
			if mask&(1<<i) != 0 {
				fields[i] = []byte{}
			}
		}
		p, s, e := CloneScannerByteFields(fields[0], fields[1], fields[2])
		for i, f := range [3][]byte{p, s, e} {
			if (f == nil) != (fields[i] == nil) || len(f) != 0 || cap(f) != 0 {
				t.Fatalf("mask=%d field=%d nil/empty changed", mask, i)
			}
		}
	}
	p, s, e := CloneScannerByteFields(nil, []byte{}, []byte("x"))
	if p != nil || s == nil || len(s) != 0 || string(e) != "x" {
		t.Fatal("mixed absent, empty and nonempty fields changed")
	}
}

var scannerFieldCloneSink [3][]byte

func TestCloneScannerByteFieldsAllocationBound(t *testing.T) {
	a, b, c := []byte("payload"), []byte("start"), []byte("end")
	allocs := testing.AllocsPerRun(100, func() { p, s, e := CloneScannerByteFields(a, b, c); scannerFieldCloneSink = [3][]byte{p, s, e} })
	if allocs > 1 {
		t.Fatalf("freezing three fields allocates %.0f objects, want at most one", allocs)
	}
}
