package parsercorephase0

// CloneScannerByteFields freezes the three scanner byte fields in one private
// allocation. Each field has a disjoint range and no spare capacity, so writes
// and appends cannot modify another field. Nil and empty fields stay distinct.
func CloneScannerByteFields(payload, start, end []byte) ([]byte, []byte, []byte) {
	const maxInt = int(^uint(0) >> 1)
	total := len(payload)
	if len(start) > maxInt-total || len(end) > maxInt-total-len(start) {
		// Keep independent allocation behavior for an unrepresentable combined
		// length rather than overflowing the allocation size.
		return cloneScannerByteField(payload), cloneScannerByteField(start), cloneScannerByteField(end)
	}
	total += len(start) + len(end)
	storage := make([]byte, total)
	var p, s, e []byte
	offset := 0
	if payload != nil {
		n := len(payload)
		p = storage[offset : offset+n : offset+n]
		copy(p, payload)
		offset += n
	}
	if start != nil {
		n := len(start)
		s = storage[offset : offset+n : offset+n]
		copy(s, start)
		offset += n
	}
	if end != nil {
		n := len(end)
		e = storage[offset : offset+n : offset+n]
		copy(e, end)
	}
	return p, s, e
}
func cloneScannerByteField(src []byte) []byte {
	if src == nil {
		return nil
	}
	out := make([]byte, len(src))
	copy(out, src)
	return out
}
