//go:build cgo && treesitter_c_parity && gts_parsercorephase0 && !gts_no_parsercorephase0
package cgoharness
import (
 "strings"
 "testing"
 gts "github.com/odvcencio/gotreesitter"
 "github.com/odvcencio/gotreesitter/grammars"
 sitter "github.com/tree-sitter/go-tree-sitter"
)
func TestGoDFASelectionOracleEvidence(t *testing.T){
 lang:=grammars.GoLanguage();cl,err:=ParityCLanguage("go");if err!=nil{t.Fatal(err)}
 for _,name:=range []string{"_expression","_type","_simple_type","expression_list","call_expression","type_conversion_expression","index_expression","generic_type","type_arguments","identifier","type_identifier"}{
  var own,cids []int
  for id,n:=range lang.SymbolNames{if n==name{own=append(own,id)}}
  for id:=uint32(0);id<cl.NodeKindCount();id++{if cl.NodeKindForId(uint16(id))==name{cids=append(cids,int(id))}}
  t.Logf("ORACLE_SYMBOL name=%s go=%v C=%v",name,own,cids)
 }
 t.Logf("ORACLE_ARTIFACT structural_election_certified=%t",lang.CompactAcceptanceStructuralElectionCertified)
 parser:=sitter.NewParser();defer parser.Close();if err:=parser.SetLanguage(cl);err!=nil{t.Fatal(err)}
 parser.SetLogger(func(kind sitter.LogType,message string){if kind==sitter.LogTypeParse && (strings.HasPrefix(message,"select_")||strings.Contains(message,"reduce sym:")){t.Logf("ORACLE_C_LOG %s",message)}})
 source:=[]byte("package p\nfunc f() { _ = F[p.T](x,y) }\n")
 tree:=parser.Parse(source,nil);if tree==nil||tree.RootNode()==nil||tree.RootNode().HasError(){t.Fatal("C rejected source")};defer tree.Close()
 t.Logf("ORACLE_C_TREE %s",tree.RootNode().ToSexp())
 prod:=gts.NewParser(lang);prod.SetAdmissionCandidateRoute(false);prod.SetGLRTrace(true);got,err:=prod.Parse(source);if err!=nil||got==nil{t.Fatal(err)};defer got.Release()
 t.Logf("ORACLE_LEGACY_TREE %s",got.RootNode().SExpr(lang))
 diffs:=compactT3StructuralDivergences(got.RootNode(),lang,tree.RootNode());if len(diffs)!=0{t.Fatalf("legacy reference diverged: %d",len(diffs))}
}
