//go:build cgo && treesitter_c_parity && gts_parsercorephase0 && !gts_no_parsercorephase0

package cgoharness

import (
	gts "github.com/odvcencio/gotreesitter"
	"github.com/odvcencio/gotreesitter/grammars"
	"testing"
)

func TestGoGenericCallFrontierLockedCProbe(t *testing.T) {
	cLang, err := ParityCLanguage("go")
	if err != nil {
		t.Fatal(err)
	}
	lang := grammars.GoLanguage()
	cases := []struct{ name, source string }{
 {"qualified_type_multi","package p\nfunc f() { _ = F[p.T](x,y) }\n"},
 {"qualified_type_empty","package p\nfunc f() { _ = F[p.T]() }\n"},
 }

	for _, tc := range cases {
		t.Run(tc.name, func(t *testing.T) {
			source := []byte(tc.source)
			ct := compactT3ParseC(t, cLang, source)
			defer ct.Close()
			if ct.RootNode().HasError() {
				t.Fatal("locked C rejected witness")
			}
			p := gts.NewParser(lang)
			p.SetAdmissionCandidateRoute(true)
			br, bf := gts.AdmissionCandidateCounters()
			tr, err := p.Parse(source)
			if err != nil || tr == nil {
				t.Fatal(err)
			}
			defer tr.Release()
			r, f := gts.AdmissionCandidateCounters()
			reason := ""
			if f > bf {
				reason = gts.AdmissionCandidateLastFallbackReason()
			}
			if diffs := compactT3StructuralDivergences(tr.RootNode(), lang, ct.RootNode()); len(diffs) != 0 {
				for _, d := range diffs[:min(len(diffs), 8)] {
					t.Error(compactT3FormatDivergence(d))
				}
				t.Fatalf("%d differences", len(diffs))
			}
			t.Logf("GENERIC_PUBLIC case=%s compact=%d fallback=%d reason=%q C_exact=true", tc.name, r-br, f-bf, reason)
		})
	}
}
