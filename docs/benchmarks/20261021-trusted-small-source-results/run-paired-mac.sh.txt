#!/usr/bin/env bash
set -euo pipefail
export PATH=/protocol/bin:/usr/local/go/bin:$PATH GOMAXPROCS=1 GOWORK=off GOFLAGS='-p=1 -mod=readonly -buildvcs=false -overlay=/protocol/benchmark-overlay.json'
cd /candidate
go version > /evidence/go-version.txt
GTS_ADMISSION_CANDIDATE=0 bash /protocol/run_randomized_benchmarks.sh --output /evidence/cogno-after.txt --baseline-root /baseline --baseline-output /evidence/cogno-before.txt --lock-path /evidence/benchmark.lock --bench-regex '^BenchmarkCognocienteGo137KiBSuite$' --require-benchmarks BenchmarkCognocienteGo137KiBSuite/Full,BenchmarkCognocienteGo137KiBSuite/InsertPlain,BenchmarkCognocienteGo137KiBSuite/InsertProfiled,BenchmarkCognocienteGo137KiBSuite/ReplacePlain,BenchmarkCognocienteGo137KiBSuite/ReplaceProfiled,BenchmarkCognocienteGo137KiBSuite/DeletePlain,BenchmarkCognocienteGo137KiBSuite/DeleteProfiled,BenchmarkCognocienteGo137KiBSuite/NoEditPlain,BenchmarkCognocienteGo137KiBSuite/NoEditProfiled --runs 20 --seed-start 131001 --benchtime 750ms > /evidence/cogno-runner.log 2>&1
GTS_ADMISSION_CANDIDATE=0 bash /protocol/run_randomized_benchmarks.sh --output /evidence/go-after.txt --baseline-root /baseline --baseline-output /evidence/go-before.txt --lock-path /evidence/benchmark.lock --bench-regex '^BenchmarkSmallSourceReceiptProductionGo$' --require-benchmarks BenchmarkSmallSourceReceiptProductionGo/full_dfa,BenchmarkSmallSourceReceiptProductionGo/single_byte_edit_dfa,BenchmarkSmallSourceReceiptProductionGo/no_edit_dfa --runs 20 --seed-start 131101 --benchtime 750ms > /evidence/go-runner.log 2>&1
bash /protocol/run_randomized_benchmarks.sh --output /evidence/yaml-after.txt --baseline-root /baseline --baseline-output /evidence/yaml-before.txt --lock-path /evidence/benchmark.lock --bench-regex '^BenchmarkCompactThroughputYAML$' --require-benchmarks BenchmarkCompactThroughputYAML/r4_legacy,BenchmarkCompactThroughputYAML/eof8k_legacy,BenchmarkCompactThroughputYAML/r4_ordinary,BenchmarkCompactThroughputYAML/eof8k_ordinary,BenchmarkCompactThroughputYAML/r4_owned,BenchmarkCompactThroughputYAML/eof8k_owned --runs 20 --seed-start 131201 --benchtime 750ms > /evidence/yaml-runner.log 2>&1
printf 'complete\n' > /evidence/complete.txt
