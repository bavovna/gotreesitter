#!/usr/bin/env bash
set -euo pipefail
export PATH=/usr/local/go/bin:$PATH GOMAXPROCS=1 GOWORK=off GOFLAGS='-p=1 -mod=readonly -buildvcs=false'
git config --global --add safe.directory /protocol
cd /protocol
bash scripts/run_randomized_benchmarks_test.sh > /evidence/driver-selftest.log 2>&1
GOTOOLCHAIN=go1.22.0 go version > /evidence/go122-version.txt
GOTOOLCHAIN=go1.22.0 bash scripts/run_primary_go_suite_test.sh > /evidence/go122-contract.log 2>&1
GOTOOLCHAIN=local go version > /evidence/go125-version.txt
GOTOOLCHAIN=local bash scripts/run_primary_go_suite_test.sh > /evidence/go125-contract.log 2>&1
printf 'complete\n' > /evidence/contracts-complete.txt
cd /candidate
GOTOOLCHAIN=local bash /protocol/scripts/run_randomized_benchmarks.sh --primary-go-suite --output /evidence/head.txt --baseline-root /baseline --baseline-output /evidence/base.txt --lock-path /evidence/benchmark.lock --runs 20 --seed-start 1 --benchtime 750ms --tags '' > /evidence/production-runner.log 2>&1
printf 'complete\n' > /evidence/complete.txt
