package gotreesitter_test

import (
	"crypto/sha256"
	"fmt"
	"github.com/odvcencio/gotreesitter/internal/benchfixtures"
	"os"
	"testing"
)

func TestWritePinnedGoMemoryWitness(t *testing.T) {
	source, _, err := benchfixtures.GeneratedSource("go", 1024*1024)
	if err != nil {
		t.Fatal(err)
	}
	if len(source) != 1048591 || fmt.Sprintf("%x", sha256.Sum256(source)) != "76c70b62127085d3c791ba98a24cba9f22bd343e40afb43522bbb5041cede524" {
		t.Fatal("witness drift")
	}
	if err := os.WriteFile("/evidence/source.go.txt", source, 0600); err != nil {
		t.Fatal(err)
	}
}
