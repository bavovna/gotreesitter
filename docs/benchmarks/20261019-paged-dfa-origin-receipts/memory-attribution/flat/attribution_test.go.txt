type memoryAttributionRow struct {
	Name                string
	Length, Capacity    int
	ElementBytes, Bytes uint64
}

func collectMemoryAttribution(value reflect.Value, prefix string, depth int) []memoryAttributionRow {
	if value.Kind() == reflect.Pointer {
		if value.IsNil() {
			return nil
		}
		value = value.Elem()
	}
	if value.Kind() != reflect.Struct {
		return nil
	}
	var rows []memoryAttributionRow
	for i := 0; i < value.NumField(); i++ {
		field := value.Field(i)
		name := prefix + value.Type().Field(i).Name
		if field.Kind() == reflect.Slice {
			size := uint64(field.Type().Elem().Size())
			rows = append(rows, memoryAttributionRow{name, field.Len(), field.Cap(), size, uint64(field.Cap()) * size})
		} else if depth > 0 && (field.Kind() == reflect.Struct || field.Kind() == reflect.Pointer) {
			rows = append(rows, collectMemoryAttribution(field, name+".", depth-1)...)
		}
	}
	return rows
}
func TestGoRuntimeMemoryAttribution(t *testing.T) {
	source, err := os.ReadFile("/evidence/source.go.txt")
	if err != nil {
		t.Fatal(err)
	}
	if len(source) != 1048591 || fmt.Sprintf("%x", sha256.Sum256(source)) != "76c70b62127085d3c791ba98a24cba9f22bd343e40afb43522bbb5041cede524" {
		t.Fatal("witness drift")
	}
	p := newAdmissionCandidateGoParser(t)
	runner, borrowed, err := p.borrowAdmissionCandidateRunner()
	if err != nil {
		t.Fatal(err)
	}
	defer p.returnAdmissionCandidateRunner(runner, borrowed)
	runtime.GC()
	var before, after runtime.MemStats
	runtime.ReadMemStats(&before)
	tree, ok, reason := p.tryCompactFullParseRoute(source)
	if !ok || tree == nil {
		t.Fatalf("compact declined: %s", reason)
	}
	defer tree.Release()
	runtime.ReadMemStats(&after)
	if tree.RootNode().HasError() || tree.RootNode().EndByte() != uint32(len(source)) || tree.ParseRuntime().NodesAllocated != 903086 || tree.ParseRuntime().TokensConsumed != 425742 {
		t.Fatalf("work/completion changed: %s", tree.ParseRuntime().Summary())
	}
	rows := collectMemoryAttribution(reflect.ValueOf(runner.compact), "core.", 2)
	rows = append(rows, collectMemoryAttribution(reflect.ValueOf(&runner.scheduler), "scheduler.", 1)...)
	rows = append(rows, collectMemoryAttribution(reflect.ValueOf(&runner.scratch), "scratch.", 1)...)
	sort.Slice(rows, func(i, j int) bool { return rows[i].Bytes > rows[j].Bytes })
	result := map[string]any{"source_bytes": len(source), "heap_before": before.HeapAlloc, "heap_after": after.HeapAlloc, "heap_inuse_after": after.HeapInuse, "allocated_bytes": after.TotalAlloc - before.TotalAlloc, "gc_cycles": after.NumGC - before.NumGC, "core_footprint": runner.compact.FootprintBytes(), "scheduler_footprint": diagnosticParserCoreSchedulerFootprintBytes(&runner.scheduler), "arena_volume": arenaAllocatedVolume(tree.arena), "core_work": runner.compact.Work(), "scheduler_work": runner.scheduler.work, "buffers": rows}
	encoded, err := json.Marshal(result)
	if err != nil {
		t.Fatal(err)
	}
	t.Logf("GO_MEMORY_ATTRIBUTION %s", encoded)
}
