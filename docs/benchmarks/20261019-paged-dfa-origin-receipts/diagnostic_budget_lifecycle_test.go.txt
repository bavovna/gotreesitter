func requirePrivateProtocolWork(t testing.TB, id string, c *core.Core) {
	t.Helper()
	b, err := json.Marshal(c.Work())
	if err != nil || string(b) != dispatchRestorePrivateWork[id] {
		t.Fatalf("private work changed for %s: %s err=%v", id, b, err)
	}
}
func preparePrivateProtocolBudgetFixture(t testing.TB) (*parserCoreWarmPrepared, error) {
	t.Helper()
	fixture := loadDiagnosticParserCoreCanonicalFixture(t, "query_compile")
	for _, row := range diagnosticParserCoreCanonicalAdmissions {
		if row.id == fixture.ID {
			requireDiagnosticParserCoreCanonicalFixtureIdentity(t, fixture, row)
		}
	}
	runner, err := newParserCoreFreshFullRunner(parserCoreWarmGoScanner, parserCoreFreshFullCanonicalOptions())
	if err != nil {
		return nil, err
	}
	scheduler, source, err := runner.executeSchedulerOpen(fixture.Source, runner.compact, true)
	if source != nil {
		source.Close()
	}
	if err != nil {
		return nil, err
	}
	requirePrivateProtocolWork(t, fixture.ID, runner.compact)
	return &parserCoreWarmPrepared{parserCoreFreshFullRunner: runner, source: fixture.Source, acceptedCore: runner.compact, acceptedHead: scheduler.acceptedHead}, nil
}
func TestDiagnosticPrivateChangedSourceKeepsRebuildBudget(t *testing.T) {
	prepared, err := preparePrivateProtocolBudgetFixture(t)
	if err != nil {
		t.Fatal(err)
	}
	paths, err := prepared.acceptedCore.Derivations(prepared.acceptedHead)
	if err != nil || len(paths) != 1 {
		t.Fatalf("paths=%v err=%v", paths, err)
	}
	var publishScratch parserCoreRunnerScratch
	tree, err := materializeDiagnosticParserCoreAcceptedSelection(prepared.acceptedCore,
		prepared.acceptedHead, paths[0].Payloads, prepared.parser, prepared.source, &publishScratch, true, false)
	if err != nil {
		t.Fatal(err)
	}
	runner := parserCoreFreshFullRunner{parser: prepared.parser, replayParseStates: true}
	if !runner.scratch.acceptanceMaterialization.Store(prepared.acceptedCore,
		prepared.acceptedHead, paths[0].Payloads, compactAcceptanceMaterializationContext(true), tree) {
		tree.Release()
		t.Fatal("store failed")
	}
	defer runner.scratch.acceptanceMaterialization.Discard()
	scheduler := diagnosticParserCoreGenericScheduler{compact: prepared.acceptedCore,
		acceptedHead: prepared.acceptedHead, acceptedPayloads: paths[0].Payloads}
	scheduler.options.stopControlMemoryBudgetBytes = 1
	result, err := runner.materializeSelection(nil, prepared.acceptedCore, &scheduler)
	if result != nil {
		result.Release()
		t.Fatal("changed source published a cached tree")
	}
	var decline *diagnosticParserCoreDecline
	if !errors.As(err, &decline) || decline.boundary != DiagnosticParserCoreCap {
		t.Fatalf("changed-source rebuild lost its budget: %v", err)
	}
	if runner.scratch.acceptanceMaterialization.Retained() != nil {
		t.Fatal("changed source retained the old publication")
	}
}

func TestDiagnosticPrivateRepeatedCanonicalLifecycle(t *testing.T) {
	runner, err := newParserCoreFreshFullRunner(parserCoreWarmGoScanner, parserCoreFreshFullCanonicalOptions())
	if err != nil {
		t.Fatal(err)
	}
	for pass := 0; pass < 2; pass++ {
		for _, row := range diagnosticParserCoreCanonicalAdmissions {
			fixture := loadDiagnosticParserCoreCanonicalFixture(t, row.id)
			requireDiagnosticParserCoreCanonicalFixtureIdentity(t, fixture, row)
			tree, err := runner.parse(fixture.Source)
			if err != nil {
				t.Fatalf("pass %d fixture %s: %v", pass, row.id, err)
			}
			requireDiagnosticParserCoreCanonicalEOF(t, tree, len(fixture.Source))
			if got := requireDiagnosticParserCoreCanonicalTreeDigest(t, tree, runner.lang); got != row.deepTreeSHA256 {
				tree.Release()
				t.Fatalf("pass %d fixture %s deep-tree digest=%s want=%s", pass, row.id, got, row.deepTreeSHA256)
			}
			requirePrivateProtocolWork(t, row.id, runner.compact)
			tree.Release()
		}
	}
}
