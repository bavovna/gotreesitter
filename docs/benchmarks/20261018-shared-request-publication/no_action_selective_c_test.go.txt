package cgoharness

import (
	"encoding/json"
	gts "github.com/odvcencio/gotreesitter"
	"github.com/odvcencio/gotreesitter/grammars"
	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
	"os"
	"testing"
)

func TestSharedRequestNoActionBoundary(t *testing.T) {
	source, err := os.ReadFile("../internal/benchfixtures/testdata/real/yaml")
	if err != nil {
		t.Fatal(err)
	}
	lang := grammars.YamlLanguage()
	cl, err := ParityCLanguage("yaml")
	if err != nil {
		t.Fatal(err)
	}
	for _, mode := range []string{"owned", "early", "late", "selective"} {
		t.Run(mode, func(t *testing.T) {
			t.Setenv("GTS_SHARED_PUBLISH_PHASE", mode)
			t.Cleanup(core.SetExternalLexerOwnershipProbeForTest(mode == "owned"))
			var finish func() []gts.ActualSharedRequestObservation
			if mode != "owned" {
				finish = gts.SeedActualSharedRequestsForTest()
			}
			capture := gts.CaptureSharedNoActionForTest()
			p := gts.NewParser(lang)
			p.SetAdmissionCandidateRoute(true)
			beforeA, beforeB := gts.AdmissionCandidateCounters()
			tree, err := p.Parse(source)
			events := capture()
			if finish != nil {
				finish()
			}
			if err != nil || tree == nil {
				t.Fatal(err)
			}
			defer tree.Release()
			ct := compactT3ParseC(t, cl, source)
			defer ct.Close()
			assertLockedCTreeExactWithErrors(t, mode, tree, lang, ct)
			a, b := gts.AdmissionCandidateCounters()
			a, b = a-beforeA, b-beforeB
			if mode == "late" || mode == "selective" {
				if a != 0 || b != 1 {
					t.Fatal("late control changed")
				}
			} else {
				if a != 1 || b != 0 {
					t.Fatal("passing control changed")
				}
			}
			raw, err := json.Marshal(struct {
				Mode            string
				Route, Fallback uint64
				Events          []gts.SharedNoActionEvent
			}{mode, a, b, events})
			if err != nil {
				t.Fatal(err)
			}
			t.Logf("NO_ACTION_BOUNDARY %s", raw)
		})
	}
}
