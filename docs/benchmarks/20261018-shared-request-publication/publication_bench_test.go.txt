//go:build gts_parsercorephase0

package gotreesitter_test

import (
	"bytes"
	gts "github.com/odvcencio/gotreesitter"
	"github.com/odvcencio/gotreesitter/grammars"
	core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
	"os"
	"testing"
)

func sharedPublicationBenchmarkSetup(b *testing.B) {
	b.Helper()
	cwd, err := os.Getwd()
	if err != nil {
		b.Fatal(err)
	}
	role := "1"
	if cwd == "/baseline" {
		role = "0"
	}
	b.Setenv("GTS_SHARED_ATOMIC_PUBLICATION", role)
	b.Cleanup(core.SetExternalLexerOwnershipProbeForTest(false))
	finish := gts.SeedActualSharedRequestsForBenchmark()
	b.Cleanup(func() { finish() })
}
func BenchmarkSharedRequestPublicationR4(b *testing.B) {
	sharedPublicationBenchmarkSetup(b)
	benchmarkAdmissionR4(b, true)
	if a, c := gts.AdmissionCandidateCounters(); a != uint64(b.N) || c != 0 {
		b.Fatalf("routed=%d fallback=%d", a, c)
	}
}
func BenchmarkSharedRequestPublication8KiB(b *testing.B) {
	sharedPublicationBenchmarkSetup(b)
	sample, err := os.ReadFile("internal/benchfixtures/testdata/real/yaml")
	if err != nil {
		b.Fatal(err)
	}
	doc := append([]byte("---\n"), sample...)
	doc = append(doc, '\n')
	source := bytes.Repeat(doc, 4)
	p := gts.NewParser(grammars.YamlLanguage())
	p.SetAdmissionCandidateRoute(true)
	parse := func() {
		tree, err := p.Parse(source)
		if err != nil || tree == nil {
			b.Fatal(err)
		}
		if tree.RootNode().HasError() || tree.RootNode().EndByte() != uint32(len(source)) {
			b.Fatal("incomplete tree")
		}
		tree.Release()
	}
	parse()
	gts.ResetAdmissionCandidateCountersForTest()
	b.ReportAllocs()
	b.SetBytes(int64(len(source)))
	b.ResetTimer()
	for range b.N {
		parse()
	}
	b.StopTimer()
	a, c := gts.AdmissionCandidateCounters()
	if a != uint64(b.N) || c != 0 {
		b.Fatalf("routed=%d fallback=%d", a, c)
	}
	b.ReportMetric(float64(a)/float64(b.N), "compact-routes/op")
	b.ReportMetric(float64(c)/float64(b.N), "compact-fallbacks/op")
}
func BenchmarkSharedRequestPublicationLegacyControl(b *testing.B) { benchmarkAdmissionR4(b, false) }
