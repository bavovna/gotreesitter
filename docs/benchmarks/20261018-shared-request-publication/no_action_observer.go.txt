//go:build gts_parsercorephase0

package gotreesitter

import core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"

type SharedNoActionRequest struct {
	Reference             uint32
	Origin                uint64
	State                 StateID
	Token                 Token
	BeforeByte, AfterByte int
	BeforeID, AfterID     core.CheckpointID
	Valid, AfterMatches   bool
	Election              int
}
type SharedNoActionHeader struct {
	Sequence                                                uint64
	State                                                   core.StateID
	Byte                                                    uint32
	Shifted, EffectivelyShifted, Paused, Accepted, Reopened bool
	RequestReference                                        uint32
	BaseByte                                                int
	Requests                                                []SharedNoActionRequest
}
type SharedNoActionEvent struct {
	Election      int
	Token         Token
	DropIndices   []int
	EpochProgress bool
	Headers       []SharedNoActionHeader
}

var sharedNoActionObserver func(*diagnosticParserCoreGenericScheduler, []int)

func diagnosticSharedNoActionObserve(s *diagnosticParserCoreGenericScheduler, indices []int) {
	if sharedNoActionObserver != nil {
		sharedNoActionObserver(s, indices)
	}
}
func CaptureSharedNoActionForTest() func() []SharedNoActionEvent {
	prior := sharedNoActionObserver
	var events []SharedNoActionEvent
	sharedNoActionObserver = func(s *diagnosticParserCoreGenericScheduler, indices []int) {
		e := SharedNoActionEvent{Election: s.electionIndex, Token: s.token, DropIndices: append([]int(nil), indices...), EpochProgress: s.epochProgress}
		for _, h := range s.headers {
			state, offset, err := s.compact.Boundary(h.head)
			if err != nil {
				panic(err)
			}
			row := SharedNoActionHeader{Sequence: h.creationSeq, State: state, Byte: offset, Shifted: h.shifted, EffectivelyShifted: h.effectivelyShifted(), Paused: h.paused, Accepted: h.accepted, Reopened: h.isZeroWidthReopened(), RequestReference: h.versionLexerRequestReference(), BaseByte: -1}
			if b := h.versionLexerSnapshot(); b != nil {
				row.BaseByte = b.dfa.lexerPos
			}
			for i, r := range s.versionLexerRequests {
				v := SharedNoActionRequest{Reference: uint32(i + 1), Origin: r.headerCreationSeq, State: r.state, Token: r.token, BeforeByte: -1, AfterByte: -1, BeforeID: r.beforeID, AfterID: r.afterID, Valid: r.valid, Election: r.electionIndex}
				if r.before != nil {
					v.BeforeByte = r.before.dfa.lexerPos
				}
				if r.after != nil {
					v.AfterByte = r.after.dfa.lexerPos
				}
				v.AfterMatches = diagnosticParserCoreVersionLexerSnapshotEqual(r.after, h.versionLexerSnapshot())
				row.Requests = append(row.Requests, v)
			}
			e.Headers = append(e.Headers, row)
		}
		events = append(events, e)
	}
	return func() []SharedNoActionEvent { sharedNoActionObserver = prior; return events }
}
