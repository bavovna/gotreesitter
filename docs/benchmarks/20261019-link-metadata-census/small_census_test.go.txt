//go:build gts_parsercorephase0
package gotreesitter_test
import (
 "bytes"
 "crypto/sha256"
 "os"
 "testing"
 gts "github.com/odvcencio/gotreesitter"
 "github.com/odvcencio/gotreesitter/grammars"
 "github.com/odvcencio/gotreesitter/internal/benchfixtures"
 core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
)
func TestLinkMetadataSmallCensus(t *testing.T) {
 name := os.Getenv("GTS_LINK_METADATA_CENSUS")
 var source []byte
 var language *gts.Language
 switch name {
 case "go_query", "go_grammargen":
  id := "query_compile"; if name == "go_grammargen" { id = "grammargen_lr" }
  for _, fixture := range benchfixtures.GoFullParseFixtures() { if fixture.ID == id { var err error; source,err=fixture.Load();if err!=nil{t.Fatal(err)} } }
  language=grammars.GoLanguage()
 case "yaml_r4_owned", "yaml_8k_owned":
  sample,err:=os.ReadFile("internal/benchfixtures/testdata/real/yaml");if err!=nil{t.Fatal(err)}
  source=sample
  if name=="yaml_8k_owned" { document:=append([]byte("---\n"),sample...);document=append(document,'\n');source=bytes.Repeat(document,4) }
  language=grammars.YamlLanguage();t.Cleanup(core.SetExternalLexerOwnershipProbeForTest(true))
 default: t.Fatal("unknown census case")
 }
 if len(source)==0 {t.Fatal("missing source")}
 parser:=gts.NewParser(language);parser.SetAdmissionCandidateRoute(true)
 br,bf:=gts.AdmissionCandidateCounters();tree,err:=parser.Parse(source)
 if err!=nil || tree==nil {t.Fatalf("parse: %v",err)};defer tree.Release()
 cr,cf:=gts.AdmissionCandidateCounters();if cr-br!=1 || cf-bf!=0 {t.Fatal("census did not use compact")}
 if tree.RootNode().HasError() || tree.RootNode().EndByte()!=uint32(len(source)) {t.Fatal("incomplete census parse")}
 rt:=tree.ParseRuntime();t.Logf("source=%s bytes=%d sha256=%x nodes=%d tokens=%d",name,len(source),sha256.Sum256(source),rt.NodesAllocated,rt.TokensConsumed)
}
