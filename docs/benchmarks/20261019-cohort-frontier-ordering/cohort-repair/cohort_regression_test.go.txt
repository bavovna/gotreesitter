//go:build gts_parsercorephase0
package gotreesitter
import (
 "testing"
 core "github.com/odvcencio/gotreesitter/internal/parsercorephase0"
)
func TestOrdinaryShiftCohortPreservesAlreadyShiftedSibling(t *testing.T){
 table:=&genericConflictTable{cells:map[genericConflictCell][]core.Action{
  {state:5,symbol:9}:{{Type:core.ActionShift,State:8}},
  {state:6,symbol:9}:{{Type:core.ActionShift,State:8}},
  {state:7,symbol:9}:{{Type:core.ActionShift,State:8}},
 }}
 compact,err:=core.New(table,core.Limits{});if err!=nil{t.Fatal(err)}
 first,err:=compact.Seed(5,10);if err!=nil{t.Fatal(err)}
 second,err:=compact.Seed(6,10);if err!=nil{t.Fatal(err)}
 third,err:=compact.Seed(7,10);if err!=nil{t.Fatal(err)}
 early,err:=compact.Shift(third,9,0,core.Token{Symbol:9,StartByte:10,EndByte:20},core.ForkOrder{});if err!=nil{t.Fatal(err)}
 earlyPaths,err:=compact.Derivations(early);if err!=nil||len(earlyPaths)!=1||len(earlyPaths[0].Payloads)!=1{t.Fatalf("early paths=%+v err=%v",earlyPaths,err)}
 earlyPayload:=earlyPaths[0].Payloads[0]
 token:=Token{Symbol:9,StartByte:10,EndByte:20}
 scheduler:=&diagnosticParserCoreGenericScheduler{
  compact:compact,headers:[]diagnosticParserCoreHeader{{head:first,creationSeq:1},{head:second,creationSeq:2},{head:early,creationSeq:3,shifted:true}},
  token:token,currentElection:DiagnosticParserCoreElection{Token:token},options:DiagnosticParserCorePrefixOptions{MaxDispatches:10},receipt:&DiagnosticParserCoreGenericScheduler{},
 }
 stop,err:=scheduler.dispatchPass();if err!=nil||stop!=nil{t.Fatalf("dispatch stop=%+v err=%v",stop,err)}
 if len(scheduler.headers)!=1||!scheduler.headers[0].shifted{t.Fatalf("frontier=%+v, want one merged shifted head",scheduler.headers)}
 paths,err:=compact.Derivations(scheduler.headers[0].head);if err!=nil||len(paths)!=3{t.Fatalf("paths=%+v err=%v, want all three incoming histories",paths,err)}
 found:=false
 for _,path:=range paths{if len(path.Payloads)==1&&path.Payloads[0]==earlyPayload{found=true}}
 if !found{t.Fatal("cohort lost the already-shifted conflict arm")}
}
