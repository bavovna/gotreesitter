//go:build cgo && treesitter_c_parity && gts_parsercorephase0 && !gts_no_parsercorephase0
package cgoharness
import (
 "fmt"
 "testing"
 gts "github.com/odvcencio/gotreesitter"
 "github.com/odvcencio/gotreesitter/grammars"
)
func TestGoTokenTurnExpandedLockedCProbe(t *testing.T){
 cLang,err:=ParityCLanguage("go");if err!=nil{t.Fatal(err)}
 lang:=grammars.GoLanguage()
 names:=[]string{"F","p.G"}
 types:=[]string{"T","int","*T","[]T","map[string]T","p.T","T,U","func(int) int","struct{ X int }","chan T"}
 args:=[]string{"x","x,y","F[T](x)","p.G[T](x)",""}
 contexts:=[]string{"_ = %s","return %s","if true { _ = %s }"}
 for ni,name:=range names{for ti,ty:=range types{for ai,arg:=range args{for ci,context:=range contexts{
  caseName:=fmt.Sprintf("n%d_t%d_a%d_c%d",ni,ti,ai,ci)
  t.Run(caseName,func(t *testing.T){
   source:=[]byte("package p\nfunc f() { "+fmt.Sprintf(context,name+"["+ty+"]("+arg+")")+" }\n")
   ct:=compactT3ParseC(t,cLang,source);defer ct.Close()
   if ct.RootNode().HasError(){t.Fatalf("C rejected generated witness: %s",source)}
   p:=gts.NewParser(lang);p.SetAdmissionCandidateRoute(true)
   r0,f0:=gts.AdmissionCandidateCounters();tree,err:=p.Parse(source)
   if err!=nil||tree==nil{t.Fatal(err)};defer tree.Release()
   r1,f1:=gts.AdmissionCandidateCounters()
   t.Logf("EXPANDED_ROUTE compact=%d fallback=%d source=%q",r1-r0,f1-f0,source)
   assertLockedCTreeExact(t,"token-turn "+caseName,tree,lang,ct)
  })
 }}}}
}
