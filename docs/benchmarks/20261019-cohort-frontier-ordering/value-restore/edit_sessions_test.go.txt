//go:build cgo && treesitter_c_parity && gts_parsercorephase0 && !gts_no_parsercorephase0
package cgoharness
import (
 "fmt"
 "testing"
 gts "github.com/odvcencio/gotreesitter"
 "github.com/odvcencio/gotreesitter/grammars"
)
func TestGoVersionTurnEditSessionsLockedC(t *testing.T){
 cl,err:=ParityCLanguage("go");if err!=nil{t.Fatal(err)};lang:=grammars.GoLanguage()
 for ni,name:=range []string{"F","p.G"}{for ti,ty:=range []string{"T","int","*T","p.T"}{for ci,context:=range []string{"_ = %s","return %s","if true { _ = %s }"}{
  t.Run(fmt.Sprintf("n%d_t%d_c%d",ni,ti,ci),func(t *testing.T){
   p:=gts.NewParser(lang);p.SetAdmissionCandidateRoute(true)
   var tree *gts.Tree;var before []byte
   defer func(){if tree!=nil{tree.Release()}}()
   for step,args:=range []string{"x","x,y","","z","F[T](x)","x"}{
    source:=[]byte("package p\nfunc f() { "+fmt.Sprintf(context,name+"["+ty+"]("+args+")")+" }\n")
    if tree==nil {tree,err=p.Parse(source)} else {
     start:=0;for start<len(before)&&start<len(source)&&before[start]==source[start]{start++}
     tail:=0;for tail<len(before)-start&&tail<len(source)-start&&before[len(before)-1-tail]==source[len(source)-1-tail]{tail++}
     oldEnd,newEnd:=len(before)-tail,len(source)-tail
     tree.Edit(gts.InputEdit{StartByte:uint32(start),OldEndByte:uint32(oldEnd),NewEndByte:uint32(newEnd),StartPoint:pointAtOffset(before,start),OldEndPoint:pointAtOffset(before,oldEnd),NewEndPoint:pointAtOffset(source,newEnd)})
     old:=tree;tree,err=p.ParseIncremental(source,old);if tree==old{t.Fatal("edited parse returned the old tree")};old.Release()
    }
    if err!=nil||tree==nil{t.Fatalf("step %d incremental parse: %v",step,err)}
    cp:=compactT3ParseC(t,cl,source)
    if cp.RootNode().HasError(){cp.Close();t.Fatalf("C rejected step %d: %s",step,source)}
    freshParser:=gts.NewParser(lang);freshParser.SetAdmissionCandidateRoute(true);fresh,freshErr:=freshParser.Parse(source)
    if freshErr!=nil||fresh==nil{cp.Close();t.Fatal(freshErr)}
    assertLockedCTreeExact(t,fmt.Sprintf("step %d incremental",step),tree,lang,cp)
    assertLockedCTreeExact(t,fmt.Sprintf("step %d fresh",step),fresh,lang,cp)
    fresh.Release();cp.Close()
    var noEditErr error;var changed bool
    allocations:=testing.AllocsPerRun(10,func(){same,e:=p.ParseIncremental(source,tree);noEditErr=e;changed=changed||same!=tree})
    if allocations!=0||noEditErr!=nil||changed{t.Fatalf("step %d no-edit allocations=%g err=%v changed=%t",step,allocations,noEditErr,changed)}
    before=source
   }
  })
 }}}
}
