func TestPrototypeDFATerminalSourceRejectsChangedOrigin(t *testing.T) {
 runner,err:=newParserCoreFreshFullRunner(parserCoreWarmGoScanner,parserCoreFreshFullCanonicalOptions());if err!=nil{t.Fatal(err)}
 checked:=false
 var retained *prototypeDFATerminalSource
 observer:=diagnosticParserCoreSeedObserver{afterElection:func(s *diagnosticParserCoreGenericScheduler)(bool,error){
  if checked || !s.token.lexerInternalDFALexed(){return false,nil}
  checked=true
  doneToken,doneActive:=s.beginPrototypeDFADispatch(s.token,true);defer s.endPrototypeDFADispatch(doneToken,doneActive)
  p:=&s.dfaIdentitySource;retained=p
  token:=core.Token{Symbol:core.Symbol(s.token.Symbol),StartByte:s.token.StartByte,EndByte:s.token.EndByte,LexerSkippedPrefixLength:diagnosticParserCoreLexerSkippedPrefixLength(s.token,s.options.captureLexerSkippedPrefixProvenance)}
  if !p.authenticatesSharedDFATerminal(token){t.Fatal("actual shared DFA source was not authenticated")}
  for _,name:=range []string{"inactive","different keyword","different read frontier","input copy","input length","different language","different lexer","different symbol","different end","external","different skipped prefix"}{
   q:=*p;tok:=token
   switch name{
   case "inactive":q.active=false
   case "different keyword":q.dispatch.lexFlags^=tokenFlagKeyword
   case "different read frontier":q.dispatch.lexerLookaheadEndByte++
   case "input copy":q.input=append([]byte(nil),q.input...)
   case "input length":q.input=q.input[:len(q.input)-1]
   case "different language":q.language=&Language{}
   case "different lexer":q.lexer=&Lexer{}
   case "different symbol":tok.Symbol++
   case "different end":tok.EndByte++
   case "external":tok.External=true
   case "different skipped prefix":tok.LexerSkippedPrefixLength++
   }
   if q.authenticatesSharedDFATerminal(tok){t.Fatalf("%s changed ownership but remained authenticated",name)}
  }
  for _,name:=range []string{"unshared request","not internally lexed","missing","no lookahead","external token","error lex mode","error retry"}{
   oldToken,oldElection:=s.token,s.currentElection.Token
   shared:=true
   switch name{
   case "unshared request":shared=false
   case "not internally lexed":s.token.lexFlags &^=tokenFlagInternalDFALexed
   case "missing":s.token.Missing=true
   case "no lookahead":s.token.NoLookahead=true
   case "external token":s.token.ExternalScannerToken=true
   case "error lex mode":s.token.lexFlags|=tokenFlagErrorModeLexed
   case "error retry":s.token.lexFlags|=tokenFlagErrorModeRetried
   }
   s.currentElection.Token=s.token
   undoToken,undoActive:=s.beginPrototypeDFADispatch(s.token,shared)
   if p.authenticatesSharedDFATerminal(token){t.Errorf("%s was certified",name)}
   s.endPrototypeDFADispatch(undoToken,undoActive);s.token,s.currentElection.Token=oldToken,oldElection
  }
  owner:=s.freshSessionOwner;s.freshSessionOwner=nil
  if p.authenticatesSharedDFATerminal(token){t.Error("source outlived its owner")};s.freshSessionOwner=owner
  return false,nil
 }}
 source:=[]byte("package p\nfunc f() { _ = F(x) }\n")
 _,tokens,err:=runner.executeSchedulerOpenWithObserver(source,runner.compact,false,observer)
 if tokens!=nil{tokens.Close()}
 if err!=nil{t.Fatal(err)}
 if !checked{t.Fatal("no actual internal DFA election exercised")}
 if retained==nil || retained.scheduler!=nil || retained.source!=nil || retained.input!=nil{t.Fatal("fresh run retained a source identity")}
}

func TestPrototypeDFADispatchRestoresNestedPanicWithoutAllocating(t *testing.T){
 s:=&diagnosticParserCoreGenericScheduler{}
 before:=Token{Symbol:7,StartByte:10,EndByte:12}
 s.dfaIdentitySource.dispatch,s.dfaIdentitySource.active=before,true
 run:=func(){
  saved,active:=s.beginPrototypeDFADispatch(Token{Symbol:8},false)
  defer s.endPrototypeDFADispatch(saved,active)
 }
 if a:=testing.AllocsPerRun(100,run);a!=0{t.Fatalf("dispatch restore allocated %g times",a)}
 sentinel:=new(int)
 func(){
  defer func(){if got:=recover();got!=sentinel{t.Fatalf("panic=%v, want sentinel",got)}}()
  saved,active:=s.beginPrototypeDFADispatch(Token{Symbol:8},false)
  defer s.endPrototypeDFADispatch(saved,active)
  func(){
   nested,nestedActive:=s.beginPrototypeDFADispatch(Token{Symbol:9},false)
   defer s.endPrototypeDFADispatch(nested,nestedActive)
   panic(sentinel)
  }()
 }()
 if s.dfaIdentitySource.dispatch!=before||!s.dfaIdentitySource.active{t.Fatal("nested panic did not restore complete prior context")}
}
